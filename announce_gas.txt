// ① テスト送信：紐付けマスタ先頭ユーザー（=自分想定）に1通
function testScheduleAnnouncementToMe() {
  var token = PropertiesService.getScriptProperties().getProperty('LINE_CHANNEL_ACCESS_TOKEN');
  if (!token) { console.error('LINE_CHANNEL_ACCESS_TOKEN 未設定'); return; }
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = ss.getSheetByName('紐付けマスタ').getDataRange().getValues();
  if (data.length < 2) { console.error('紐付けマスタにレコードがありません'); return; }
  var userId = data[1][data[0].indexOf('LINE_userId')];
  var res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ to: userId, messages: [buildScheduleAnnouncementFlex()] }),
    muteHttpExceptions: true
  });
  console.log('test push →', userId, '/', res.getResponseCode(), res.getContentText());
}

// ② 一斉配信：全友だちへ broadcast
// ⚠️ 実行＝即送信・月間通数を消費。必ず①を先に実行して見た目を確認すること
function broadcastScheduleAnnouncement() {
  var token = PropertiesService.getScriptProperties().getProperty('LINE_CHANNEL_ACCESS_TOKEN');
  if (!token) { console.error('LINE_CHANNEL_ACCESS_TOKEN 未設定'); return; }
  var res = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/broadcast', {
    method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({ messages: [buildScheduleAnnouncementFlex()] }),
    muteHttpExceptions: true
  });
  console.log('broadcast →', res.getResponseCode(), res.getContentText());
}
